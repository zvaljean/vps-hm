            inherit inputs;
          };
          modules =
            [
              {
                home.username = username;
                home.homeDirectory = homeDirectory;
                home.stateVersion = "26.05";
                programs.home-manager.enable = true;
              }
              ./modules/home
              ./modules/pkgs
              ./modules/home/trigger.nix
            ]
            ++ extraModules;
        };
    in {
      # add formatter
      # formatter.${system} = pkgs.alejandra;

      homeConfigurations = {
        data = mkHome {
          username = "data";
          homeDirectory = "/home/data";
        };
        root = mkHome {
          username = "root";
          homeDirectory = "/root";
          extraModules = [

            })
          ];
        };
      };
    };
  }
#+end_src

- 重新 tangle 后，在对应用户的登录环境执行 =home-manager switch --flake .#data= 或 =home-manager switch --flake .#root=。
- 新用户通过 =mkHome= 添加；用户专属配置可放入 =extraModules=，覆盖共用值时使用 =lib.mkForce=。
- 两个入口共用 trigger；data 默认启用 frps 并监听 10001，root 通过 extraModules 关闭 frps，避免同机端口冲突。

  # https://nix-community.github.io/home-manager/index.xhtml#sec-module-auto-importing
  { lib, ... }:
  let
    dir = builtins.readDir ./.;
    autoImportFilter = name: type:
      let
        isValidFile = type == "regular" && name != "default.nix" && lib.strings.hasSuffix ".nix" name;
        isValidDir = type == "directory";
        isNotIgnored = !(lib.strings.hasPrefix "_" name);
      in
        isNotIgnored && (isValidFile || isValidDir);

    validModules = builtins.filter (name: autoImportFilter name dir.${name}) (builtins.attrNames dir);
  in
  {
    imports = map (name: ./${name}) validModules;
  }

      set -eu
      umask 077
      forwarded_socket="$HOME/.gnupg/S.gpg-agent"
      [ -S "$forwarded_socket" ] || exit 0

      agent_socket=$(${pkgs.gnupg}/bin/gpgconf --list-dirs agent-socket)
      [ -n "$agent_socket" ] || exit 0
      [ "$agent_socket" != "$forwarded_socket" ] || exit 0
      [ ! "$agent_socket" -ef "$forwarded_socket" ] || exit 0

      ${pkgs.coreutils}/bin/mkdir -p "$(${pkgs.coreutils}/bin/dirname "$agent_socket")"
      # 锁文件保持存在，避免删除锁文件后不同进程锁住不同 inode。
      exec 9>"$agent_socket.forward.lock"
      ${pkgs.util-linux}/bin/flock -n 9 || exit 0

      # 其他 shell 可能已经完成接管，拿到锁后必须重新检查。
      [ -S "$forwarded_socket" ] || exit 0
      [ ! "$agent_socket" -ef "$forwarded_socket" ] || exit 0
      if [ -S "$agent_socket" ] && [ ! -L "$agent_socket" ]; then

      # 不覆盖普通文件、目录，或停止失败后仍占据路径的本地 socket。
      if [ -L "$agent_socket" ] || [ ! -e "$agent_socket" ]; then
